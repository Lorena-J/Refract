\section{La pregunta}
\begin{frame}{Empieza por la pregunta}
\begin{block}{Pregunta de investigación}
¿Qué cambia al rediseñar un proceso de investigación?
\end{block}
\vspace{4mm}
\begin{itemize}\setlength{\itemsep}{5mm}
\item \textbf{Contexto:} Explica el problema y a quién afecta.
\item \textbf{Laguna:} Indica qué dejan sin resolver los enfoques actuales.
\item \textbf{Aportación:} Describe las nuevas pruebas o el método que ofreces.
\end{itemize}
\end{frame}

\begin{frame}{Ofrece una ruta al público}
\begin{columns}[T,onlytextwidth]
\column{0.30\textwidth}
{\Huge\color{accent}01}\par\vspace{3mm}
\textbf{Pregunta}\par\vspace{2mm}
¿Qué importa y qué necesitamos saber?
\column{0.30\textwidth}
{\Huge\color{accent}02}\par\vspace{3mm}
\textbf{Pruebas}\par\vspace{2mm}
¿Cómo se diseñó el estudio y qué cambió?
\column{0.30\textwidth}
{\Huge\color{accent}03}\par\vspace{3mm}
\textbf{Implicaciones}\par\vspace{2mm}
¿Qué concluimos y cuál es el siguiente paso?
\end{columns}
\end{frame}
