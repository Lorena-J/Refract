# Licences for seed files the TeX Live 2022 database does not describe (scripts/tex-seed.py).
# One line each: file <bundle file name> <licence>... | glob <name pattern> <licence>... | package <name> <licence>...
# A file line wins over a glob line, which wins over the package's licence in the database.
# Licence ids are the keys of TEXTS in scripts/tex-seed.py. Each entry says where the
# licence comes from; "in-file" means the file's own header states the terms and the
# file is shipped with that header unmodified.

# hyph-utf8 pattern files: each states its licence (MIT, LPPL, GPL, LGPL, BSD or a
# plain permission to copy) in its header; the database has no licence for most
# hyphen-* packages. hyph-ro.tex ("[None]"), hyph-mn-cyrl-x-lmc.tex ("NONE"),
# hyph-en-us.tex, hyph-nb/nn/no.tex and the hyph-quote-* files need a maintainer review.
glob hyph-*.tex in-file
glob loadhyph-*.tex in-file
# Werner Lemberg's dehyph-exptl: "You may freely use, modify and/or distribute this file."
glob dehyph?-x-*.tex in-file
glob dehyph?-x-*.pat in-file

# Plain TeX's hyphen.tex: "Unlimited copying and redistribution of this file are permitted
# as long as this file is not modified" (header).
file hyphen.tex in-file
# Bernard Gaulle's placeholder files: no licence statement in the file; hyphen-base.
file dumyhyph.tex other-free
file zerohyph.tex other-free
# language.dat: "Created long ago by Sebastian Rahtz and others. Public domain." (header).
file language.dat pd
# Greek patterns for Ibycus encoding, translated from hyph-grc (LPPL, see hyph-grc.tex).
file ibyhyph.tex lppl1.3c

# Adobe Glyph List 2.0: Apache-2.0 (header). pdfglyphlist.txt is compiled from it.
file glyphlist.txt apache2
file pdfglyphlist.txt apache2
# lcdf-typetools texglyphlist.txt: lcdf-typetools is GPL 2 or later.
file texglyphlist.txt gpl2+
# Font map files generated by updmap from the installed map files: a list of font names.
file pdftex.map other-free
file kanjix.map other-free
# pgf: "under the LaTeX Project Public License and/or under the GNU Public License" (header).
file pgfkeysfiltered.code.tex lppl1.3c
# Written by Tectonic for its own format file (MIT, the Tectonic licence).
file tectonic-format-latex.tex mit
